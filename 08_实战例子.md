## 例 1：用纯 NumPy + Python 写一个 PID 仿真 + 画图

```python
"""
pid_sim.py
一个 PID 控制一阶系统的仿真，输出 CSV + 画图。
"""
import numpy as np
import matplotlib.pyplot as plt


class PID:
    def __init__(self, kp, ki, kd, output_max=10.0):
        self.kp, self.ki, self.kd = kp, ki, kd
        self.output_max = output_max
        self.integral = 0.0
        self.prev_err = 0.0

    def reset(self):
        self.integral = 0.0
        self.prev_err = 0.0

    def update(self, target, current, dt):
        err = target - current
        self.integral += err * dt
        deriv = (err - self.prev_err) / dt
        self.prev_err = err
        u = self.kp*err + self.ki*self.integral + self.kd*deriv
        return float(np.clip(u, -self.output_max, self.output_max))


class FirstOrderPlant:
    """一阶系统：tau * dy/dt = -y + u"""
    def __init__(self, tau=0.5):
        self.tau = tau
        self.y = 0.0

    def step(self, u, dt):
        self.y += (-self.y + u) / self.tau * dt
        return self.y


def simulate(kp, ki, kd, target_func, T=5.0, dt=0.01):
    pid = PID(kp, ki, kd)
    plant = FirstOrderPlant(tau=0.5)

    times = np.arange(0, T, dt)
    targets = np.array([target_func(t) for t in times])
    ys = np.zeros_like(times)
    us = np.zeros_like(times)

    for i, t in enumerate(times):
        u = pid.update(targets[i], plant.y, dt)
        plant.step(u, dt)
        ys[i] = plant.y
        us[i] = u

    return times, targets, ys, us


def square_wave(t):
    return 1.0 if (t % 2.0) < 1.0 else -1.0


if __name__ == "__main__":
    times, targets, ys, us = simulate(
        kp=2.0, ki=0.5, kd=0.05,
        target_func=square_wave,
        T=8.0, dt=0.01
    )

    fig, axes = plt.subplots(2, 1, sharex=True, figsize=(10, 6))
    axes[0].plot(times, targets, '--', label='target')
    axes[0].plot(times, ys, label='output')
    axes[0].legend(); axes[0].set_ylabel('y')
    axes[0].set_title(f'PID step response (kp=2, ki=0.5, kd=0.05)')

    axes[1].plot(times, us, color='orange')
    axes[1].set_xlabel('t (s)'); axes[1].set_ylabel('u')

    plt.tight_layout()
    plt.savefig('pid_response.png', dpi=120)
    print("已保存 pid_response.png")
```

## 例 2：用 PyTorch 训练一个最小 RL 策略（CartPole + REINFORCE）

```python
"""
reinforce_cartpole.py
用最简的策略梯度方法（REINFORCE）训练 CartPole-v1。
不依赖 stable-baselines / rsl_rl，纯手写，看清 RL 到底怎么训。
"""
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import gymnasium as gym


class PolicyNet(nn.Module):
    def __init__(self, obs_dim, act_dim, hidden=64):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, hidden), nn.Tanh(),
            nn.Linear(hidden, hidden), nn.Tanh(),
            nn.Linear(hidden, act_dim),
        )

    def forward(self, x):
        return self.net(x)        # logits


def run_episode(env, policy, device):
    obs, _ = env.reset()
    log_probs = []
    rewards = []
    while True:
        obs_t = torch.tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
        logits = policy(obs_t)
        dist = torch.distributions.Categorical(logits=logits)
        action = dist.sample()
        log_prob = dist.log_prob(action)

        obs, r, term, trunc, _ = env.step(action.item())
        log_probs.append(log_prob)
        rewards.append(r)
        if term or trunc:
            break
    return log_probs, rewards


def compute_returns(rewards, gamma=0.99):
    """折扣回报 G_t = r_t + γ r_{t+1} + γ² r_{t+2} + ..."""
    returns = []
    G = 0
    for r in reversed(rewards):
        G = r + gamma * G
        returns.insert(0, G)
    returns = torch.tensor(returns, dtype=torch.float32)
    # 标准化（减少方差）
    returns = (returns - returns.mean()) / (returns.std() + 1e-8)
    return returns


def train():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    env = gym.make("CartPole-v1")
    obs_dim = env.observation_space.shape[0]   # 4
    act_dim = env.action_space.n               # 2
    policy = PolicyNet(obs_dim, act_dim).to(device)
    optimizer = optim.Adam(policy.parameters(), lr=1e-2)

    for epi in range(500):
        log_probs, rewards = run_episode(env, policy, device)
        returns = compute_returns(rewards).to(device)

        log_probs_t = torch.cat(log_probs)
        loss = -(log_probs_t * returns).sum()      # 策略梯度损失

        optimizer.zero_grad()
        loss.backward()
        optimizer.step()

        if epi % 20 == 0:
            print(f"episode {epi}: 步数={len(rewards):3d}  loss={loss.item():.2f}")

    env.close()
    torch.save(policy.state_dict(), "cartpole_policy.pt")
    print("已保存 cartpole_policy.pt")


if __name__ == "__main__":
    train()
```

- 概率分布采样（`torch.distributions`）
- log probability、回报计算

## 例 3：加载训练好的策略 + 渲染

```python
"""
play_cartpole.py
加载 cartpole_policy.pt 并可视化运行。
"""
import torch
import torch.nn as nn
import gymnasium as gym


class PolicyNet(nn.Module):
    def __init__(self, obs_dim, act_dim, hidden=64):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(obs_dim, hidden), nn.Tanh(),
            nn.Linear(hidden, hidden), nn.Tanh(),
            nn.Linear(hidden, act_dim),
        )
    def forward(self, x):
        return self.net(x)


env = gym.make("CartPole-v1", render_mode="human")
policy = PolicyNet(4, 2)
policy.load_state_dict(torch.load("cartpole_policy.pt"))
policy.eval()

for ep in range(5):
    obs, _ = env.reset()
    total = 0
    while True:
        with torch.no_grad():
            logits = policy(torch.tensor(obs, dtype=torch.float32).unsqueeze(0))
            action = logits.argmax(dim=-1).item()    # 部署时取最优动作
        obs, r, term, trunc, _ = env.step(action)
        total += r
        if term or trunc:
            print(f"ep {ep}: 步数 {total}")
            break
env.close()
```

> 💡 这就是 **sim2real 部署的雏形**：保存 → 加载 → 推理 → 喂动作。
> 实物上你的"环境"换成"机器人 + IMU 读数"，"动作"换成"电机命令"。

## 例 4：批处理 / 向量化（连接 Isaac Lab 思维）

来一个对比，让你感受 Isaac Lab 思维：

**单个环境（慢）：**
```python
for env in envs:                  # 4096 个 env，循环
    obs = env.get_obs()
    action = policy(obs)
    env.step(action)
```

**向量化（快，Isaac Lab 风格）：**
```python
obs_batch = scene.get_obs()              # shape (4096, obs_dim)
action_batch = policy(obs_batch)          # 一次前向，shape (4096, act_dim)
scene.step(action_batch)                  # 4096 个 env 一起 step
```

**关键**：所有数据保持 `(num_envs, ...)` 的形状，所有运算用 PyTorch / NumPy 的批量 API。

实战代码：
```python
import torch

# 假装是 4096 个机器人的当前速度
curr_vel = torch.randn(4096, 3, device="cuda")
# 命令速度，每个机器人不一样
cmd_vel = torch.randn(4096, 3, device="cuda")

# 一行算 4096 个速度跟踪奖励
err_sq = ((cmd_vel - curr_vel) ** 2).sum(dim=1)        # (4096,)
reward = torch.exp(-err_sq / 0.25)                     # (4096,)

print(reward.shape, reward.mean().item())
```

## 例 5：把 PID 类向量化（PyTorch）

让 PID 一次控制 4096 个电机：

```python
import torch

class BatchedPID:
    def __init__(self, num_envs, kp, ki, kd, output_max, device="cuda"):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.output_max = output_max
        self.integral = torch.zeros(num_envs, device=device)
        self.prev_err = torch.zeros(num_envs, device=device)

    def update(self, target, current, dt):
        err = target - current
        self.integral += err * dt
        deriv = (err - self.prev_err) / dt
        self.prev_err = err
        u = self.kp*err + self.ki*self.integral + self.kd*deriv
        return torch.clamp(u, -self.output_max, self.output_max)


# 用法
pid = BatchedPID(num_envs=4096, kp=2.0, ki=0.1, kd=0.05, output_max=10.0)

target = torch.full((4096,), 5.0, device="cuda")
current = torch.zeros(4096, device="cuda")
u = pid.update(target, current, dt=0.01)
print(u.shape)        # torch.Size([4096])
```

> 💡 体会一下：单个 PID 的逻辑没变，只是把 float 换成了 (4096,) 的 tensor。**这就是 Isaac Lab / GPU RL 的标准风格**。

## 例 7：Sim2Real 部署的 Python 端

```python
"""
deploy.py
Sim2Real 端的策略推理（在 PC 或机载电脑上跑）。
通常和 ROS / 串口通信代码搭配。
"""
import torch
import numpy as np

# 加载训练好的策略
class Policy(torch.nn.Module):
    def __init__(self, obs_dim, act_dim):
        super().__init__()
        self.net = torch.nn.Sequential(
            torch.nn.Linear(obs_dim, 128), torch.nn.ELU(),
            torch.nn.Linear(128, 128), torch.nn.ELU(),
            torch.nn.Linear(128, act_dim),
        )
    def forward(self, x):
        return self.net(x)


policy = Policy(obs_dim=48, act_dim=12)
policy.load_state_dict(torch.load("policy.pt", map_location="cpu"))
policy.eval()

# 控制循环
@torch.no_grad()
def step_policy(obs_np: np.ndarray) -> np.ndarray:
    """obs_np: (48,)，返回动作 (12,)"""
    obs_t = torch.from_numpy(obs_np.astype(np.float32)).unsqueeze(0)
    action = policy(obs_t).squeeze(0).numpy()
    return action


# 主循环（伪代码）
# while True:
#     obs = read_from_robot()       # 你的机器人通信
#     action = step_policy(obs)
#     send_to_robot(action)
#     sleep(1/50)                    # 50Hz
```

部署时几个工程关键点：
- `model.eval()` + `with torch.no_grad()`：关掉训练特有逻辑
- 输入 normalization 必须**和训练时一致**
- 控制频率：观察 → 推理 → 输出 → 等下一个 tick
- 安全：边界保护、死区、急停（这些在 C++ 控制板里实现）
